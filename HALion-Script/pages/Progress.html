Progress

The properties of the Progress class can be used to monitor the progress of asynchronous operations, such as loading elements of VST presets or processing functions like analyzeWavetableAsync.

On this page:

Progress Class

Classes

Progress Class

Description

Functions such as loadPresetAsync, appendLayerAsync, insertLayerAsync and analyzeWavetableAsync return a Progress object. The properties of the Progress object are described by the following fields.

Available in: Controller.

Fields

FieldDescriptionValue Type
.progressThe progress in the range from 0 to 1.number
.rootThe result of the completed operation. For loading functions, this is the Element object of the first element (root) of the loaded VST preset, which can be of the type Layer, Program,or Instance. For processing functions like analyzeWavetableAsync, this is the processed object, e.g. the analyzed Wavetable Table.Layer, Program, Instance, or Wavetable Table
.cancelSet this to true to cancel the operation.boolean
.errorMessage if an error occurred.string
.infoUser definable field, for example, to manage several threads.string or table

Example for Loading Functions

-- Start with an empty program, remove any existing layers.

layers = this.parent:findLayers()
 
if layers then
   for i, layer in ipairs(layers) do
       this.parent:removeLayer(layer)
   end
end
 
-- Table with layer presets from Skylab.
layerPresets = {
   { name = "Ambient Pad 01", path = "vstsound://724ACB205EFF46F885735D1B216C37AD/.AppData/Steinberg/Skylab/Sub Presets/Layer Presets/Ambient Pads/Ambient Pad 01.vstpreset" },
   { name = "Ambient Pad 02", path = "vstsound://724ACB205EFF46F885735D1B216C37AD/.AppData/Steinberg/Skylab/Sub Presets/Layer Presets/Ambient Pads/Ambient Pad 02.vstpreset" },
   { name = "Ambient Pad 03", path = "vstsound://724ACB205EFF46F885735D1B216C37AD/.AppData/Steinberg/Skylab/Sub Presets/Layer Presets/Ambient Pads/Ambient Pad 03.vstpreset" },
   { name = "Ambient Pad 04", path = "vstsound://724ACB205EFF46F885735D1B216C37AD/.AppData/Steinberg/Skylab/Sub Presets/Layer Presets/Ambient Pads/Ambient Pad 04.vstpreset" },
}
-- Create a table with the preset names.
function getPresetNames()
   presetNames = {}
   for i, preset in ipairs(layerPresets) do
       presetNames[i] = preset.name
   end
end
 
getPresetNames()
 
-- Remove the old layer after the new one was added.
function removeOldLayer(progressInfo)
   local newPreset = progressInfo.root
   if oldPreset then
       this.parent:removeLayer(oldPreset)
       print(oldPreset.name.." removed.")
   end
   oldPreset = newPreset
end
 
-- Append the preset in a separate thread.
function appendNewLayer(progressInfo)
   if progressInfo.root then
       this.parent:appendLayerAsync(progressInfo.root, removeOldLayer)
       print("Appending "..progressInfo.root.name.."...")
   end
end
 
-- Load the preset in a separate thread.
function onSelectPresetChanged()
   progress = 0
   progressInf = loadPresetAsync(layerPresets[SelectPreset].path, appendNewLayer)
   print("Loading "..layerPresets[SelectPreset].name.."...")
end
 
-- Define a parameter for selecting the preset to be loaded.
defineParameter("SelectPreset", "Select Preset", 1, presetNames, onSelectPresetChanged)
 
-- Monitor the load progress with onIdle.
progress = 1
function onIdle()
   if progress < 1 then
       progress = progressInf.progress
       print("Progress: "..(progressInf.progress * 100).."%")
   end
end

Example for Processing Functions

defineParameter { name = "StartProcess", default = false, onChanged = function() if StartProcess then onStart() end end }
defineParameter { name = "CancelProcess", default = false}
defineParameter { name = "WavetableStart", default = 0, min = 0, max = 1024, increment = 1 }
defineParameter { name = "WavetableLength", default = 0, min = 0, max = 1024, increment = 1 }



function onStart()
    zones = this.parent:findZones(true)
    for i, zone in ipairs(zones) do
        local function onFinished(progressInf)
            if progressInf then
                if progressInf.error then
                    print(progressInf.error)
                elseif not progressInf.cancel then
                    print("Progress: 100%")
                    local wt = progressInf.root
                    if wt then
                        print("Wavetable "..wt.name.." analyzed.")
                        zone:setWavetable(1, wt)
                    end
                end
            end
        end
        local wt = zone:getWavetable (1)
        if wt then
            if WavetableStart > 0 then
                if WavetableLength > 1 then
                    local waves = {}
                    local startIdx = math.min (WavetableStart, #wt)
                    local endIdx = math.min (WavetableStart + WavetableLength - 1, #wt)
                    for idx = startIdx, endIdx do table.insert(waves, idx) end
                    print(string.format("Analyze Waves %d to %d of Wavetable ", startIdx, endIdx)..wt.name)
                    progressInf = analyzeWavetableAsync(wt, waves, onFinished)
                elseif WavetableLength == 1 then
                    local wave = math.min (WavetableStart, #wt)
                    print(string.format("Analyze Wave %d of Wavetable ", wave)..wt.name)
                    progressInf = analyzeWavetableAsync(wt, wave, onFinished)
                else
                    print("Nothing to analyze of Wavetable "..wt.name)
                end
            else
                print("Analyze Wavetable "..wt.name)
                progressInf = analyzeWavetableAsync (wt, nil, onFinished)
            end
            while progressInf.progress < 1 do
                if CancelProcess then
                    progressInf.cancel = true
                    break
                elseif progressInf.error then
                    break
                else
                    local progressPercent = 100 * progressInf.progress
                    print(string.format("Progress: %2d%%", progressPercent))
                    wait(2000)
                end
            end
            if CancelProcess then
                CancelProcess = false
                print("Wavetable analysis canceled!")
                break
            end
        end
    end
    print("Wavetable analysis finished.")
    StartProcess = false
end
